"use client";
import React, { useState, useMemo } from 'react';
import { useRouter } from 'next/navigation';
import { 
  Search, Filter, MapPin, Star, Shield, Briefcase, 
  MessageSquare, ChevronDown, CheckCircle2, X, SlidersHorizontal, 
  User, Mail, Phone, LayoutGrid, List, Zap, Heart, Share2
} from 'lucide-react';
import { useQuery } from "convex/react";
import { api } from "../../../convex/_generated/api";
import { Id } from "../../../convex/_generated/dataModel";

// --- Types & Interfaces ---

interface Lawyer {
  _id: Id<"lawyers">;
  userId: Id<"users">;
  name: string;
  email: string;
  image: string;
  phone: string;
  bio: string;
  specializations: string[];
  yearsOfExperience: number;
  rating: number;
  availableNow: boolean;
}

// --- Utility Components ---

const FilterBadge = ({ label, active, onClick }: { label: string, active: boolean, onClick: () => void }) => (
  <button
    onClick={onClick}
    className={`px-3 py-1.5 rounded-full text-xs font-semibold transition-all border ${
      active 
      ? 'bg-black text-white border-nyayak-slate' 
      : 'bg-[#fbfbfb] text-[#111111] border-gray-200 hover:border-nyayak-slate hover:bg-[#fbfbfb]'
    }`}
  >
    {label}
  </button>
);

// --- Main Page Component ---

export default function FindLawyersPage() {
  const router = useRouter();  
  const [viewMode, setViewMode] = useState<'Grid' | 'List'>('Grid');
  const [selectedLawyer, setSelectedLawyer] = useState<Lawyer | null>(null);
  
  // Search & Filters
  const [searchQuery, setSearchQuery] = useState("");
  const [activeSpecialties, setActiveSpecialties] = useState<string[]>([]);
  const [showFilters, setShowFilters] = useState(false);
  const [onlyAvailable, setOnlyAvailable] = useState(false);

  // Fetch Data
  const lawyersData = useQuery(api.lawyers.list, { 
    specialization: activeSpecialties.length === 1 ? activeSpecialties[0] : undefined 
  });

  const isLoading = lawyersData === undefined;
  
  // Client-side filtering for complex cases (multi-select search, text search)
  const filteredLawyers = useMemo(() => {
    if (!lawyersData) return [];
    
    return lawyersData.filter(l => {
      // Text Search
      const matchSearch = l.name.toLowerCase().includes(searchQuery.toLowerCase()) || 
                          (l.bio && l.bio.toLowerCase().includes(searchQuery.toLowerCase()));
      
      // Specialization (Handle multi-select which API might not support directly)
      // If API handles single spec in args, we just double check here or handle the 'All' case
      const matchSpecialty = activeSpecialties.length === 0 || 
                             l.specializations.some(s => activeSpecialties.some(active => s.toLowerCase().includes(active.toLowerCase())));

      // Availability
      const matchAvailability = onlyAvailable ? l.availableNow : true;

      return matchSearch && matchSpecialty && matchAvailability;
    });
  }, [lawyersData, searchQuery, activeSpecialties, onlyAvailable]);

  const toggleSpecialty = (s: string) => {
    setActiveSpecialties(prev => prev.includes(s) ? prev.filter(x => x !== s) : [...prev, s]);
  };

  const allSpecialties = ['Criminal', 'Corporate', 'Family', 'Constitutional', 'IP', 'Real Estate', 'Labor', 'Tax', 'Immigration'];

  return (
    <main className="min-h-screen bg-[#F3EEE6] pt-24 pb-12 relative overflow-hidden flex flex-col items-center">
      {/* Background Pattern */}
      

      {/* Main Container */}
      <div className="relative z-10 w-full max-w-7xl px-6 border border-gray-200 min-h-[calc(100vh-6rem)] mb-20 rounded-3xl bg-[#fbfbfb]/50 backdrop-blur-[2px]">
        
        <div className="py-12">
          
          {/* Header Section */}
          <div className="text-center mb-12">
            <h1 className="text-4xl font-bold text-[#111111] mb-4 tracking-tight">Legal Professionals Directory</h1>
            <p className="text-lg text-[#111111] max-w-2xl mx-auto">
              Connect with Nepal's top legal minds. Verified credentials and trusted by community.
            </p>
          </div>

          {/* AI Matching Banner */}
          <div className="bg-linear-to-r from-[#2c1810] to-nyayak-slate rounded-2xl p-6 md:p-8 mb-10 text-white shadow-xl relative overflow-hidden group">
            <div className="absolute right-0 top-0 w-64 h-64 bg-[#fbfbfb]/5 rounded-full blur-3xl -mr-16 -mt-16 group-hover:bg-[#fbfbfb]/10 transition-colors duration-700"></div>
            <div className="relative z-10 flex flex-col md:flex-row items-center justify-between gap-6">
              <div>
                <div className="flex items-center gap-2 mb-2 text-nyayak-mute font-semibold text-sm uppercase tracking-wider">
                  <Zap size={16} className="text-yellow-400" /> AI-Powered Matching
                  <span className="bg-yellow-500/20 text-yellow-300 text-[10px] px-2 py-0.5 rounded-full border border-yellow-500/30">Coming Soon</span>
                </div>
                <h2 className="text-2xl font-bold mb-2">Want to hire with AI precision?</h2>
                <p className="text-gray-500 max-w-lg opacity-80">
                  Provide details about your case, and our AI will summarize your needs to match you with the 3 most qualified lawyers tailored specifically to your legal situation.
                </p>
              </div>
              <button 
                disabled
                className="px-6 py-3 bg-[#fbfbfb]/20 text-white/50 font-bold rounded-xl shadow-none cursor-not-allowed flex items-center gap-2 whitespace-nowrap"
              >
                <Search size={18} /> Find My Match
              </button>
            </div>
          </div>

          {/* Main Content Areas */}
          <div className="flex flex-col lg:flex-row gap-8">
            
            {/* Sidebar Filters */}
            <div className={`lg:w-80 shrink-0 space-y-6 ${showFilters ? 'block' : 'hidden lg:block'}`}>
              
              {/* Filter Card */}
              <div className="bg-[#fbfbfb] rounded-2xl border border-gray-200 p-6 shadow-sm sticky top-24">
                <div className="flex items-center justify-between mb-6">
                  <h3 className="font-bold text-[#111111] flex items-center gap-2">
                    <Filter size={18} /> Filters
                  </h3>
                  <button 
                    onClick={() => {setActiveSpecialties([]); setOnlyAvailable(false);}}
                    className="text-xs text-[#111111] hover:text-black font-medium"
                  >
                    Reset All
                  </button>
                </div>

                {/* Specialties */}
                <div className="mb-6">
                  <label className="text-sm font-semibold text-[#111111] mb-3 block">Specialization</label>
                  <div className="flex flex-wrap gap-2">
                    {allSpecialties.map((s) => (
                      <FilterBadge 
                        key={s} 
                        label={s} 
                        active={activeSpecialties.includes(s)} 
                        onClick={() => toggleSpecialty(s)} 
                      />
                    ))}
                  </div>
                </div>

                {/* Availability */}
                <div className="mb-2 border-t border-gray-200 pt-6">
                  <label className="text-sm font-semibold text-[#111111] mb-3 block">Availability</label>
                  <label className="flex items-center gap-3 cursor-pointer group">
                    <div 
                      className={`w-5 h-5 rounded border flex items-center justify-center transition-colors ${onlyAvailable ? 'bg-black border-nyayak-slate' : 'border-nyayak-slate group-hover:border-nyayak-slate'}`}
                      onClick={() => setOnlyAvailable(!onlyAvailable)}
                    >
                      {onlyAvailable && <CheckCircle2 size={12} className="text-white" />}
                    </div>
                    <span className="text-sm text-[#111111] group-hover:text-[#111111] transition-colors">Available Now Only</span>
                  </label>
                </div>
              </div>
            </div>

            {/* Results Column */}
            <div className="flex-1 min-w-0">
               {/* Search & Sort Bar */}
               <div className="bg-[#fbfbfb] rounded-2xl border border-gray-200 p-4 mb-6 flex flex-col sm:flex-row items-center justify-between gap-4 shadow-sm sticky top-2 z-10">
                  <div className="relative flex-1 w-full">
                     <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-500" size={18} />
                     <input 
                        type="text" 
                        placeholder="Search by name..." 
                        value={searchQuery}
                        onChange={(e) => setSearchQuery(e.target.value)}
                        className="w-full pl-10 pr-4 py-2 bg-[#fbfbfb] rounded-lg border border-gray-200 focus:border-nyayak-slate focus:ring-1 focus:ring-nyayak-slate outline-none transition-all text-sm"
                     />
                  </div>
                  <div className="flex items-center gap-2 w-full sm:w-auto overflow-x-auto">
                     <button 
                        onClick={() => setShowFilters(!showFilters)}
                        className="lg:hidden p-2 border border-gray-200 rounded-lg hover:bg-[#fbfbfb] text-gray-500"
                     >
                        <SlidersHorizontal size={18} />
                     </button>
                     <div className="h-6 w-px bg-nyayak-mute hidden sm:block"></div>
                     <div className="flex bg-[#fbfbfb] p-1 rounded-lg border border-gray-200">
                        <button 
                          onClick={() => setViewMode('Grid')}
                          className={`p-1.5 rounded-md transition-all ${viewMode === 'Grid' ? 'bg-[#fbfbfb] shadow text-black' : 'text-gray-500 hover:text-[#111111]'}`}
                        >
                           <LayoutGrid size={18} />
                        </button>
                        <button 
                          onClick={() => setViewMode('List')}
                          className={`p-1.5 rounded-md transition-all ${viewMode === 'List' ? 'bg-[#fbfbfb] shadow text-black' : 'text-gray-500 hover:text-[#111111]'}`}
                        >
                           <List size={18} />
                        </button>
                     </div>
                  </div>
               </div>

               {/* Loading State */}
               {isLoading && (
                  <div className="text-center py-20">
                    <div className="animate-spin w-10 h-10 border-4 border-nyayak-slate border-t-transparent rounded-full mx-auto mb-4"></div>
                    <p className="text-gray-500">Loading lawyers...</p>
                  </div>
               )}

               {/* Results Grid */}
               {!isLoading && filteredLawyers.length === 0 ? (
                  <div className="text-center py-20 bg-[#fbfbfb] rounded-3xl border border-dashed border-gray-200">
                     <div className="w-20 h-20 bg-[#fbfbfb] rounded-full flex items-center justify-center mx-auto mb-4">
                        <User size={32} className="text-nyayak-mute" />
                     </div>
                     <h3 className="text-xl font-bold text-gray-500">No lawyers found</h3>
                     <p className="text-[#111111] mt-2">Try adjusting your filters or search terms.</p>
                  </div>
               ) : (
                  <div className={`grid gap-8 ${viewMode === 'Grid' ? 'grid-cols-1 md:grid-cols-2 lg:grid-cols-2 xl:grid-cols-3' : 'grid-cols-1'}`}>
                     {filteredLawyers.map((lawyer) => (
                        <div 
                           key={lawyer._id}
                           onClick={() => setSelectedLawyer(lawyer)}
                           className={`group relative bg-[#fbfbfb] rounded-[24px] border border-[#E8E1D9] p-6 shadow-sm hover:shadow-[0_8px_30px_rgba(90,58,26,0.08)] hover:border-[#D4C3B0] transition-all duration-300 cursor-pointer flex flex-col ${viewMode === 'List' ? 'md:flex-row md:items-center md:gap-8' : ''}`}
                        >
                           {/* Header */}
                           <div className={`flex items-start justify-between mb-6 ${viewMode === 'List' ? 'md:mb-0 md:w-80 md:shrink-0' : ''}`}>
                              <div className="flex items-start gap-4">
                                 <div className="relative shrink-0">
                                    <div className="w-[72px] h-[72px] rounded-2xl overflow-hidden border border-gray-200 bg-[#fbfbfb]">
                                       {lawyer.image ? (
                                           <img 
                                              src={lawyer.image} 
                                              alt={lawyer.name} 
                                              className="w-full h-full object-cover" 
                                           />
                                       ) : (
                                           <div className="w-full h-full flex items-center justify-center bg-nyayak-mute text-[#111111] font-bold text-2xl">
                                             {lawyer.name.charAt(0)}
                                           </div>
                                       )}
                                    </div>
                                    <div className={`absolute -bottom-1 -right-1 p-[3px] bg-[#fbfbfb] rounded-full`}>
                                       {lawyer.availableNow ? (
                                           <CheckCircle2 size={16} className="text-green-500 fill-white" /> 
                                       ) : (
                                           <div className="w-4 h-4 bg-gray-300 rounded-full border-2 border-white"></div>
                                       )}
                                    </div>
                                 </div>
                                 <div className="pt-1">
                                    <h3 className="font-bold text-lg text-[#111111] group-hover:text-black transition-colors line-clamp-1 leading-tight">{lawyer.name}</h3>
                                    <p className="text-sm text-[#111111] mt-1 font-medium">Legal Professional</p>
                                 </div>
                              </div>
                              <button className="text-nyayak-mute hover:text-red-500 transition-colors p-2 hover:bg-red-50 rounded-full">
                                 <Heart size={20} />
                              </button>
                           </div>

                           <div className={`space-y-6 flex-1 ${viewMode === 'List' ? 'md:space-y-0 md:grid md:grid-cols-2 md:gap-8' : ''}`}>
                              
                              {/* Stats Layout */}
                              <div className="flex items-center gap-0 border border-gray-200 rounded-xl overflow-hidden bg-[#fbfbfb]">
                                 <div className="flex-1 py-3 px-2 text-center border-r border-gray-200 hover:bg-[#fbfbfb]/50 transition-colors">
                                    <div className="flex items-center justify-center gap-1.5 font-bold text-[#111111] text-sm">
                                       <Star size={14} className="fill-yellow-400 text-yellow-400" />
                                       {lawyer.rating.toFixed(1)}
                                    </div>
                                    <div className="text-[10px] font-semibold text-gray-500 uppercase tracking-wide mt-1">Rating</div>
                                 </div>
                                 <div className="flex-1 py-3 px-2 text-center border-r border-gray-200 hover:bg-[#fbfbfb]/50 transition-colors">
                                    <div className="font-bold text-[#111111] text-sm">{lawyer.yearsOfExperience}+</div>
                                    <div className="text-[10px] font-semibold text-gray-500 uppercase tracking-wide mt-1">Years Exp.</div>
                                 </div>
                                 <div className="flex-1 py-3 px-2 text-center hover:bg-[#fbfbfb]/50 transition-colors">
                                    <div className={`font-bold text-sm ${lawyer.availableNow ? 'text-green-600' : 'text-gray-500'}`}>
                                      {lawyer.availableNow ? 'Yes' : 'No'}
                                    </div>
                                    <div className="text-[10px] font-semibold text-gray-500 uppercase tracking-wide mt-1">Available</div>
                                 </div>
                              </div>
                              
                              {/* Tags & Action - Combined in grid mode or spaced in card mode */}
                              <div className={`flex flex-col gap-4 ${viewMode === 'List' ? 'md:flex-row md:items-center md:h-full' : ''}`}>
                                 <div className={`flex flex-wrap gap-2 ${viewMode === 'List' ? '' : 'min-h-[32px]'}`}>
                                    {(lawyer.specializations || []).slice(0, 3).map(s => (
                                       <span key={s} className="px-3 py-1.5 bg-[#fbfbfb] border border-gray-200 rounded-lg text-xs text-[#111111] font-semibold tracking-wide capitalize">
                                          {s}
                                       </span>
                                    ))}
                                    {(lawyer.specializations || []).length > 3 && (
                                       <span className="px-3 py-1.5 bg-[#fbfbfb] rounded-lg text-xs text-[#111111] font-medium">+{lawyer.specializations.length - 3}</span>
                                    )}
                                 </div>

                                 <div className={`flex items-end justify-end ${viewMode === 'List' ? 'md:mt-0 md:ml-auto' : 'pt-2 border-t border-[#E8E1D9]/50'}`}>
                                    <button className="w-full md:w-auto px-5 py-2.5 bg-[#4A3222] text-white text-sm font-bold rounded-xl shadow-lg shadow-[#4A3222]/10 hover:shadow-[#4A3222]/20 hover:scale-[1.02] active:scale-95 transition-all flex items-center justify-center gap-2 group/btn">
                                       View Profile <ChevronDown size={14} className="-rotate-90 group-hover/btn:translate-x-0.5 transition-transform" strokeWidth={3} />
                                    </button>
                                 </div>
                              </div>
                           </div>
                        </div>
                     ))}
                  </div>
               )}
            </div>
          </div>
        </div>
      </div>

      {/* --- Lawyer Profile Overlay --- */}
      {selectedLawyer && (
        <div className="fixed inset-0 z-50 flex justify-end">
          <div 
             className="absolute inset-0 bg-black/40 backdrop-blur-[3px] transition-opacity" 
             onClick={() => setSelectedLawyer(null)}
          ></div>
          <div className="relative w-full max-w-3xl bg-[#fbfbfb] h-full shadow-2xl animate-in slide-in-from-right duration-300 flex flex-col overflow-hidden">
             
             {/* Header Image Background */}
             <div className="h-48 bg-linear-to-r from-[#2c1810] to-nyayak-slate relative">
                <div className="absolute inset-0 opacity-20" style={{ backgroundImage: 'url("https://www.transparenttextures.com/patterns/cubes.png")' }}></div>
                <button 
                  onClick={() => setSelectedLawyer(null)}
                  className="absolute top-4 right-4 p-2 bg-black/20 text-white rounded-full hover:bg-black/40 transition-colors backdrop-blur-sm"
                >
                   <X size={24} />
                </button>
             </div>

             {/* Profile Info Pull-up */}
             <div className="px-8 flex-1 overflow-y-auto -mt-16 relative z-10">
                <div className="flex items-end justify-between mb-6">
                   <div className="relative">
                      <div className="w-32 h-32 rounded-2xl overflow-hidden border-4 border-white shadow-xl bg-[#fbfbfb]">
                         <img src={selectedLawyer.image || `https://ui-avatars.com/api/?name=${encodeURIComponent(selectedLawyer.name)}`} alt={selectedLawyer.name} className="w-full h-full object-cover" />
                      </div>
                      <div className={`absolute -bottom-2 -right-2 ${selectedLawyer.availableNow ? 'bg-green-500' : 'bg-gray-400'} text-white px-3 py-1 rounded-full text-xs font-bold border-2 border-white shadow-sm flex items-center gap-1`}>
                         <span className="w-1.5 h-1.5 rounded-full bg-[#fbfbfb] animate-pulse"></span> {selectedLawyer.availableNow ? 'Available Now' : 'Offline'}
                      </div>
                   </div>
                   <div className="flex gap-2 mb-2">
                      <button className="p-3 bg-[#fbfbfb] border border-gray-200 rounded-xl shadow-sm hover:bg-[#fbfbfb] text-[#111111] transition-all">
                         <Share2 size={20} />
                      </button>
                   </div>
                </div>

                <div className="mb-8">
                   <h2 className="text-3xl font-bold text-[#111111] mb-1 flex items-center gap-2">
                      {selectedLawyer.name}
                   </h2>
                   <p className="text-xl text-[#111111] mb-2">Legal Professional</p>
                   <div className="flex flex-wrap gap-4 text-sm text-[#111111] mb-6">
                      <span className="flex items-center gap-1.5 text-yellow-600 font-medium"><Star size={16} className="fill-yellow-600" /> {selectedLawyer.rating.toFixed(1)} Rating</span>
                   </div>

                   {/* Tags */}
                   <div className="flex flex-wrap gap-2 mb-8">
                      {(selectedLawyer.specializations || []).map(s => (
                         <span key={s} className="px-3 py-1 bg-nyayak-mute text-[#111111] rounded-full text-sm font-medium border border-gray-200">
                            {s}
                         </span>
                      ))}
                   </div>

                   {/* Bio Section */}
                   <div className="space-y-8">
                      <section>
                         <h3 className="font-bold text-lg text-[#111111] mb-3 flex items-center gap-2"><User size={20} className="text-black" /> About</h3>
                         <p className="text-[#111111] leading-relaxed bg-[#fbfbfb] p-5 rounded-2xl border border-gray-200">
                            {selectedLawyer.bio || "No biography provided."}
                         </p>
                      </section>

                      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                         <section>
                            <h3 className="font-bold text-lg text-[#111111] mb-3 flex items-center gap-2"><Briefcase size={20} className="text-black" /> Experience</h3>
                            <div className="bg-[#fbfbfb] border border-gray-200 rounded-xl p-4 shadow-sm">
                               <div className="flex justify-between items-center mb-2">
                                  <span className="text-gray-500">Years Active</span>
                                  <span className="font-bold text-gray-500">{selectedLawyer.yearsOfExperience} Years</span>
                               </div>
                            </div>
                         </section>

                         <section>
                           <h3 className="font-bold text-lg text-[#111111] mb-3 flex items-center gap-2"><Phone size={20} className="text-black" /> Contact</h3>
                           <div className="bg-[#fbfbfb] border border-gray-200 rounded-xl p-4 shadow-sm space-y-2">
                               {selectedLawyer.email && (
                                   <div className="flex items-center gap-2 text-sm text-gray-500">
                                       <Mail size={16} className="text-gray-500" /> {selectedLawyer.email}
                                   </div>
                               )}
                               {selectedLawyer.phone && (
                                   <div className="flex items-center gap-2 text-sm text-gray-500">
                                       <Phone size={16} className="text-gray-500" /> {selectedLawyer.phone}
                                   </div>
                               )}
                           </div>
                         </section>
                      </div>
                   </div>
                </div>
             </div>

             {/* Booking Footer */}
             <div className="p-6 border-t border-gray-200 bg-[#fbfbfb] sticky bottom-0 flex gap-4 shadow-[0_-4px_6px_-1px_rgba(0,0,0,0.05)]">
                <button 
                  disabled={!selectedLawyer.availableNow}
                  className={`flex-1 py-4 text-white rounded-xl font-bold shadow-lg transition-all transform active:scale-[0.98] flex items-center justify-center gap-2 text-lg ${selectedLawyer.availableNow ? 'bg-black hover:bg-black shadow-nyayak-slate/20' : 'bg-gray-400 cursor-not-allowed'}`}
                >
                   {selectedLawyer.availableNow ? 'Book Consultation' : 'Currently Unavailable'}
                </button>
             </div>
          </div>
        </div>
      )}

    </main>
  );
}
